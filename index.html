import io import json import re import openpyxl import pandas as pd import pypdf import streamlit as st # Initialize OpenAI client (Ensure API key is set in Streamlit secrets or environment) client = OpenAI() # ============================================================================== # 1. FULL TEXT EXTRACTION # ============================================================================== def extract_full_pdf_text(pdf_file) -> str: """Scans and extracts text from ALL pages of the provided PDF file.""" try: reader = pypdf.PdfReader(pdf_file) full_text = [] for i, page in enumerate(reader.pages): text = page.extract_text() if text: full_text.append(f"--- PAGE {i+1} ---\n{text}") return "\n\n".join(full_text) except Exception as e: st.error(f"Error reading PDF {pdf_file.name}: {str(e)}") return "" # ============================================================================== # 2. CONSERVATIVE SEMANTIC CLASSIFICATION & SAFETY OVERRIDE # ============================================================================== CLASSIFICATION_PROMPT = """ You are an expert financial document classifier for wealth management systems (QPLIX). Your objective is to classify whether a document contains relevant financial, transactional, or capital accounting information. PRIORITY DIRECTIVE: - Zero False Negatives: Never exclude a document if it might contain actionable transactional data, bank transfer details, capital accounting figures, or NAV updates. - If in any doubt, classify as 'POTENTIALLY_RELEVANT'. CLASSIFICATION CATEGORIES: 1. RELEVANT: - Capital Calls / Drawdown notices - Distribution notices / Dividend payouts - Capital Account Statements / NAV reports / Valuation statements - Fee statements / Tax reporting forms - Wire/Payment instructions for investment transactions 2. POTENTIALLY_RELEVANT: - General investor letters containing embedded financial metrics - Quarterly reports with portfolio financial summaries - Transaction updates where specific amounts or dates are ambiguous - Unstructured correspondence mentioning contributions, commitments, or adjustments 3. IRRELEVANT: - Pure promotional or marketing pitch decks - General ESG policies / Legal term updates without financial metrics - KYC / AML privacy notices or boilerplate regulatory disclaimers - Event invitations or general non-financial correspondence Return ONLY a valid JSON object matching this schema: { "status": "RELEVANT" | "POTENTIALLY_RELEVANT" | "IRRELEVANT", "confidence_score": float (0.0 to 1.0), "detected_indicators": [list of strings describing specific evidence], "reasoning": "Short string explaining classification" } """ def classify_pdf(pdf_text: str, filename: str) -> dict: """Classifies PDF text using LLM and applies a conservative rule-based safety net.""" if not pdf_text.strip(): return { "status": "POTENTIALLY_RELEVANT", "confidence_score": 0.0, "detected_indicators": ["Unreadable or empty PDF"], "reasoning": "Flagged for manual review because text extraction produced no characters.", } # Primary Semantic Pass via OpenAI try: response = client.chat.completions.create( model="gpt-4o", response_format={"type": "json_object"}, messages=[ {"role": "system", "content": CLASSIFICATION_PROMPT}, { "role": "user", "content": f"Filename: {filename}\n\nDocument Text:\n{pdf_text[:12000]}", # Pass extensive context }, ], temperature=0.0, ) result = json.loads(response.choices[0].message.content) except Exception as e: result = { "status": "POTENTIALLY_RELEVANT", "confidence_score": 0.0, "detected_indicators": ["LLM processing error"], "reasoning": f"Defaulted to review due to API processing error: {str(e)}", } # Secondary Conservative Safety Net (Hard Override for Critical Indicators) # If LLM marks document as IRRELEVANT, but strong financial triggers exist, upgrade to POTENTIALLY_RELEVANT. if result["status"] == "IRRELEVANT": critical_patterns = [ r"IBAN", r"SWIFT", r"Capital Call", r"Drawdown", r"Distribution", r"Net Asset Value", r"NAV", r"Amount Due", r"Payment Instructions", r"Commitment", r"Equalisation", r"ISIN", ] text_upper = pdf_text.upper() found_triggers = [p for p in critical_patterns if re.search(r"\b" + p + r"\b", text_upper, re.IGNORECASE)] if found_triggers: result["status"] = "POTENTIALLY_RELEVANT" result["confidence_score"] = 0.5 result["detected_indicators"].append("Rule-Based Safety Override Triggered") result["reasoning"] += f" [OVERRIDE]: Document flagged because key financial term(s) {found_triggers} were found despite initial exclusion." return result # ============================================================================== # 3. QPLIX DATA EXTRACTION & TWO-STEP VALIDATION # ============================================================================== EXTRACTION_PROMPT = """ Extract all relevant QPLIX investment and financial transaction parameters from the document text. Extract the following JSON structure: { "entity_name": "string or null", "fund_name": "string or null", "transaction_type": "Capital Call" | "Distribution" | "NAV Update" | "Management Fee" | "Other" | null, "transaction_date": "YYYY-MM-DD or null", "due_date": "YYYY-MM-DD or null", "currency": "string or null", "amount": float or null, "bank_account_iban": "string or null", "notice_period_or_reference": "string or null", "additional_notes": "string or null" } """ VALIDATION_PROMPT = """ You are a Quality Control Auditor verifying data extracted from financial documents for QPLIX integration. Compare the extracted data JSON against the full document text. Perform a second pass verification to ensure: 1. No amounts, currency, due dates, or payment details were missed or misextracted. 2. If the document describes multiple amounts (e.g., Total Call vs. Share Call), ensure the correct investor-level amount was captured. 3. Verify that zero or null fields are truly absent from the source document. Return JSON format: { "is_validated": boolean, "corrected_data": { ... matching extraction schema ... }, "validation_flags": [list of warnings or missing field notes] } """ def extract_and_validate_qplix_data(pdf_text: str) -> tuple[dict, list[str]]: """Step 1: Extract structure. Step 2: Perform audit/validation check before outputting.""" # Step 1: Initial Extraction try: ext_response = client.chat.completions.create( model="gpt-4o", response_format={"type": "json_object"}, messages=[ {"role": "system", "content": EXTRACTION_PROMPT}, {"role": "user", "content": pdf_text}, ], temperature=0.0, ) extracted = json.loads(ext_response.choices[0].message.content) except Exception as e: extracted = {} # Step 2: Second Pass Validation try: val_response = client.chat.completions.create( model="gpt-4o", response_format={"type": "json_object"}, messages=[ {"role": "system", "content": VALIDATION_PROMPT}, { "role": "user", "content": f"DOCUMENT TEXT:\n{pdf_text}\n\nEXTRACTED DATA:\n{json.dumps(extracted, indent=2)}", }, ], temperature=0.0, ) val_result = json.loads(val_response.choices[0].message.content) final_data = val_result.get("corrected_data", extracted) flags = val_result.get("validation_flags", []) except Exception as e: final_data = extracted flags = [f"Validation pass failed to execute: {str(e)}"] return final_data, flags # ============================================================================== # 4. EXCEL GENERATION UTILITY # ============================================================================== def create_qplix_excel(relevant_records, review_records, excluded_records) -> bytes: """Generates structured Excel workbook with dedicated tabs for processing, review, and exclusions.""" output = io.BytesIO() with pd.ExcelWriter(output, engine="openpyxl") as writer: # Tab 1: QPLIX Import (Relevant) df_rel = pd.DataFrame(relevant_records) if df_rel.empty: df_rel = pd.DataFrame( columns=[ "Filename", "Fund Name", "Transaction Type", "Date", "Due Date", "Amount", "Currency", "IBAN", "Reference", ] ) df_rel.to_excel(writer, sheet_name="QPLIX Import Data", index=False) # Tab 2: Manual Review Needed (Borderline/Uncertain) df_rev = pd.DataFrame(review_records) if df_rev.empty: df_rev = pd.DataFrame( columns=["Filename", "Status", "Reasoning", "Flags", "Detected Indicators"] ) df_rev.to_excel(writer, sheet_name="Review Needed", index=False) # Tab 3: Excluded Documents df_exc = pd.DataFrame(excluded_records) if df_exc.empty: df_exc = pd.DataFrame( columns=["Filename", "Status", "Reasoning", "Confidence Score"] ) df_exc.to_excel(writer, sheet_name="Excluded Documents", index=False) return output.getvalue() # ============================================================================== # 5. STREAMLIT APPLICATION INTERFACE # ============================================================================== def main(): st.set_page_config( page_title="QPLIX Smart Document Filter & Extractor", layout="wide" ) st.title("QPLIX Document Processing Pipeline") st.markdown( "Upload PDF documents to automatically classify, review, and extract data for QPLIX using conservative financial relevance filtering." ) uploaded_files = st.file_uploader( "Upload PDF Files", type=["pdf"], accept_multiple_files=True ) if uploaded_files and st.button("Process Documents"): relevant_records = [] review_records = [] excluded_records = [] progress_bar = st.progress(0) for idx, file in enumerate(uploaded_files): st.text(f"Processing ({idx+1}/{len(uploaded_files)}): {file.name}") # 1. Full text scan across all pages text = extract_full_pdf_text(file) # 2. Semantic classification + safety net check classification = classify_pdf(text, file.name) status = classification["status"] if status == "RELEVANT": # 3. Extract & Perform 2nd validation check qplix_data, flags = extract_and_validate_qplix_data(text) record = {"Filename": file.name, **qplix_data} if flags: record["Validation Flags"] = "; ".join(flags) relevant_records.append(record) elif status == "POTENTIALLY_RELEVANT": # Attempt partial extraction to assist manual review qplix_data, flags = extract_and_validate_qplix_data(text) review_records.append( { "Filename": file.name, "Status": status, "Reasoning": classification.get("reasoning", ""), "Indicators": ", ".join( classification.get("detected_indicators", []) ), "Validation Warnings": "; ".join(flags), "Extracted Draft Data": json.dumps(qplix_data), } ) else: # IRRELEVANT excluded_records.append( { "Filename": file.name, "Status": status, "Reasoning": classification.get("reasoning", ""), "Confidence": classification.get("confidence_score", 0.0), } ) progress_bar.progress((idx + 1) / len(uploaded_files)) st.success("Processing Complete!") # Layout summary metrics c1, c2, c3 = st.columns(3) c1.metric("QPLIX Ready (Relevant)", len(relevant_records)) c2.metric("Flagged for Review", len(review_records)) c3.metric("Excluded", len(excluded_records)) # Excel Export excel_bytes = create_qplix_excel( relevant_records, review_records, excluded_records ) st.download_button( label="Download Structured QPLIX Excel Report", data=excel_bytes, file_name="QPLIX_Processed_Documents.xlsx", mime="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet", ) # UI Data Display st.subheader("Data Summary") if relevant_records: st.markdown("### Included Documents (QPLIX Ready)") st.dataframe(pd.DataFrame(relevant_records)) if review_records: st.markdown("### Flagged Documents (Manual Review Required)") st.dataframe(pd.DataFrame(review_records)) if excluded_records: st.markdown("### Excluded Documents") st.dataframe(pd.DataFrame(excluded_records)) if __name__ == "__main__": main()
